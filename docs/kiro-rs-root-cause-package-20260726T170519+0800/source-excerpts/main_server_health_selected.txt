  1239	async fn readyz(State(state): State<Arc<AppHealthState>>) -> impl IntoResponse {
  1240	    let postgres_ok = state.postgres_store.ping().await.is_ok();
  1241	    let redis_ok = state.redis_store.ping().await.is_ok();
  1242	    let redis_events_connected = state
  1243	        .runtime_events
  1244	        .redis_events_connected
  1245	        .load(Ordering::Acquire);
  1246	    let ready = postgres_ok && redis_ok && redis_events_connected;
  1247	    let status = if ready {
  1248	        StatusCode::OK
  1249	    } else {
  1250	        StatusCode::SERVICE_UNAVAILABLE
  1251	    };
  1252	
  1253	    (
  1254	        status,
  1255	        Json(json!({
  1256	            "status": if ready { "ready" } else { "not_ready" },
  1257	            "checks": {
  1258	                "postgres": postgres_ok,
  1259	                "redis": redis_ok,
  1260	                "redisRuntimeEvents": redis_events_connected
  1261	            },
  1262	            "lastRedisRuntimeEventAtMs": state.runtime_events.last_event_at_ms.load(Ordering::Acquire),
  1263	            "lastRedisSubscribeErrorAtMs": state.runtime_events.last_subscribe_error_at_ms.load(Ordering::Acquire)
  1264	        })),
  1265	    )

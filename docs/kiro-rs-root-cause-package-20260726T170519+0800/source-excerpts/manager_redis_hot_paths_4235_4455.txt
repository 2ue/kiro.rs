  4235	        &self,
  4236	        id: u64,
  4237	        request_weight_units: u32,
  4238	    ) -> anyhow::Result<SchedulerRedisSelectionAdmission> {
  4239	        let now = Instant::now();
  4240	        let now_ms = Utc::now().timestamp_millis();
  4241	        let rpm = {
  4242	            let global_rpm = self.config.lock().credential_rpm.unwrap_or(0);
  4243	            let entries = self.entries.lock();
  4244	            entries
  4245	                .iter()
  4246	                .find(|entry| entry.id == id)
  4247	                .map(|entry| effective_rpm(entry, global_rpm))
  4248	                .unwrap_or(global_rpm)
  4249	        };
  4250	        if rpm == 0 || self.redis_store.is_none() {
  4251	            return Ok(SchedulerRedisSelectionAdmission::NotRequired);
  4252	        }
  4253	
  4254	        let Some(redis) = &self.redis_store else {
  4255	            return Ok(SchedulerRedisSelectionAdmission::NotRequired);
  4256	        };
  4257	        let redis = redis.clone();
  4258	        let response = match self
  4259	            .await_scheduler_redis_hot_outcome("原子记录 Redis 调度选中次数", || {}, async move {
  4260	                redis
  4261	                    .try_record_scheduler_selection(id, rpm, request_weight_units)
  4262	                    .await
  4263	            })
  4264	            .await
  4265	        {
  4266	            SchedulerRedisHotOutcome::Completed(response) => response,
  4267	            SchedulerRedisHotOutcome::LocalSchedulerOverloaded => {
  4268	                return Err(SchedulerRedisUnavailableError {
  4269	                    retry_after_secs: 1,
  4270	                    local_overloaded: true,
  4271	                }
  4272	                .into());
  4273	            }
  4274	            SchedulerRedisHotOutcome::Skipped | SchedulerRedisHotOutcome::Failed { .. } => {
  4275	                return Err(SchedulerRedisUnavailableError {
  4276	                    retry_after_secs: self.scheduler_redis_retry_after_secs(),
  4277	                    local_overloaded: false,
  4278	                }
  4279	                .into());
  4280	            }
  4281	        };
  4282	
  4283	        match response {
  4284	            SchedulerSelectionReservation::Recorded {
  4285	                rate_limit_available_at_ms,
  4286	                ..
  4287	            } => {
  4288	                if let Some(available_at_ms) = rate_limit_available_at_ms
  4289	                    .and_then(|until_ms| Self::instant_from_epoch_ms(until_ms, now_ms, now))
  4290	                {
  4291	                    self.apply_rate_limit_available_at_for_credential(id, available_at_ms);
  4292	                }
  4293	                Ok(SchedulerRedisSelectionAdmission::Recorded {
  4294	                    rate_limit_available_at: rate_limit_available_at_ms
  4295	                        .and_then(|until_ms| Self::instant_from_epoch_ms(until_ms, now_ms, now)),
  4296	                })
  4297	            }
  4298	            SchedulerSelectionReservation::RateLimited {
  4299	                retry_after_ms,
  4300	                rate_limit_available_at_ms,
  4301	            } => {
  4302	                let available_at =
  4303	                    Self::instant_from_epoch_ms(rate_limit_available_at_ms, now_ms, now)
  4304	                        .unwrap_or(now + StdDuration::from_millis(retry_after_ms.max(1)));
  4305	                self.apply_rate_limit_available_at_for_credential(id, available_at);
  4306	                Ok(SchedulerRedisSelectionAdmission::RateLimited(
  4307	                    SchedulerRedisRateLimitWait {
  4308	                        retry_after: StdDuration::from_millis(retry_after_ms.max(1)),
  4309	                        available_at,
  4310	                    },
  4311	                ))
  4312	            }
  4313	        }
  4314	    }

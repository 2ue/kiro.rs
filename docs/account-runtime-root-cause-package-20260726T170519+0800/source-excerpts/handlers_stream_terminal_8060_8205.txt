  8060	                            let protocol_contamination =
  8061	                                state.ctx.response_protocol_contamination_detected();
  8062	                            if protocol_contamination && !state.downstream_committed {

  8065	                                    StreamRetryReason::ProtocolContamination,
  8066	                                    RESPONSE_PROTOCOL_CONTAMINATION_DETAIL.to_string(),
  8067	                                )
  8068	                                .await
  8069	                                {

  8071	                                        let bytes: Vec<Result<Bytes, Infallible>> = Vec::new();
  8072	                                        return Some((stream::iter(bytes), state));
  8073	                                    }

  8077	                                }

  8094	                                state.completion.report_upstream_stream_failure(format!(
  8095	                                    "{}: {}",
  8096	                                    kind, detail
  8097	                                ));
  8098	                            } else {
  8099	                                state.completion.report_success();
  8100	                            }
  8101	                            state
  8102	                                .usage_guard
  8103	                                .context()
  8104	                                .request
  8105	                                .mark_stream_terminal(if had_stream_error {
  8106	                                    if protocol_contamination {
  8107	                                        StreamTerminalReason::ProtocolContamination

  8109	                                        StreamTerminalReason::UpstreamStatusError
  8110	                                    }
  8111	                                } else {
  8112	                                    StreamTerminalReason::Completed
  8113	                                });
  8114	                            state
  8115	                                .usage_guard
  8116	                                .context()
  8117	                                .request
  8118	                                .mark_stream_events(&final_events);
  8119	                            if had_stream_error {
  8120	                                state.usage_guard.context().record_stream_failure_from_context(

  8122	                                    state.ctx.final_usage(),

  8124	                                    state.ctx.metadata_usage(),

  8131	                            state.usage_guard.complete();
  8132	                            let bytes = sse_bytes_from_events_with_initial(
  8133	                                &mut state,
  8134	                                final_events,
  8135	                                true,
  8136	                            );
  8137	                            state.finished = true;
  8138	                            Some((stream::iter(bytes), state))
  8139	                        }
  8140	                    }
  8141	                }
  8142	                _ = &mut idle_sleep => {
  8143	                    tracing::error!(
  8144	                        "上游响应流超过 {} 秒未产生数据，结束流并发送错误事件",
  8145	                        state.stream_idle_timeout_secs
  8146	                    );
  8147	                    let detail = "upstream stream idle timeout".to_string();
  8148	                    if !state.downstream_committed {
  8149	                        match retry_stream_before_downstream_commit(
  8150	                            state,
  8151	                            StreamRetryReason::IdleTimeout,
  8152	                            detail.clone(),
  8153	                        )
  8154	                        .await
  8155	                        {
  8156	                            StreamRetryOutcome::Retried(state) => {
  8157	                                let bytes: Vec<Result<Bytes, Infallible>> = Vec::new();
  8158	                                return Some((stream::iter(bytes), state));
  8159	                            }
  8160	                            StreamRetryOutcome::NotRetried(next_state) => {
  8161	                                state = next_state;

  8164	                    }
  8165	                    state.completion.report_upstream_stream_failure(detail.clone());
  8166	                    state.ctx.record_stream_error("api_error", detail);
  8167	                    let bytes = finish_stream_with_recorded_error(
  8168	                        &mut state.ctx,
  8169	                        &state.usage_guard,
  8170	                        UsageRecordStatus::UpstreamTimeout,

  8174	                    state.finished = true;

  8176	                }
  8177	                _ = state.ping_interval.tick() => {

  8179	                        let bytes: Vec<Result<Bytes, Infallible>> = Vec::new();

  8182	                    let keepalive = if claude_code_noop_delta_keepalive {

  8198	                    mark_stream_downstream_committed(&mut state);
  8199	                    let bytes: Vec<Result<Bytes, Infallible>> = vec![Ok(bytes)];
  8200	                    Some((stream::iter(bytes), state))
  8201	                }
  8202	            }
